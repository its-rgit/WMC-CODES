import os
import csv
import numpy as np
import matplotlib.pyplot as plt

WAVELENGTH_M=0.125

SAMPLE_SCAN_CSV="""Position_cm,RSSI_Indoor_dBm,RSSI_Outdoor_dBm
0,-49,-56
5,-53,-57
10,-61,-57
15,-50,-58
20,-46,-59
25,-57,-59
30,-51,-60
35,-64,-61
40,-48,-61
45,-55,-62
50,-50,-63
"""

SAMPLE_STATIONARY_CSV="""Time_s,RSSI_dBm
0,-51
5,-53
10,-50
15,-55
20,-52
25,-49
30,-54
35,-51
40,-56
45,-53
50,-50
55,-54
60,-52
"""

def load_scan(fname="scan.csv"):
    with open(fname,"w") as file:
        file.write(SAMPLE_SCAN_CSV)
    pos=[]
    indoor=[]
    outdoor=[]
    with open(fname,"r") as file:
        reader=csv.reader(file)
        for row in reader:
            if not row:
                continue
            if row[0].startswith("Position"):
                continue
            pos.append(float(row[0]))
            indoor.append(float(row[1]))
            outdoor.append(float(row[2]))
    return np.array(pos),np.array(indoor),np.array(outdoor)

def load_stationary(fname="stationary.csv"):
    with open(fname,"w") as file:
        file.write(SAMPLE_STATIONARY_CSV)
    time_s=[]
    rssi=[]
    with open(fname,"r") as file:
        reader=csv.reader(file)
        for row in reader:
            if not row:
                continue
            if row[0].startswith("Time"):
                continue
            time_s.append(float(row[0]))
            rssi.append(float(row[1]))
    return np.array(time_s),np.array(rssi)

def roughness(r):
    valid=r[~np.isnan(r)]
    if len(valid)<2:
        return 0.0
    return float(np.mean(np.abs(np.diff(valid))))

def fade_depth(r):
    valid=r[~np.isnan(r)]
    if len(valid)==0:
        return 0.0
    return float(valid.max()-valid.min())

def reversals(r):
    valid=r[~np.isnan(r)]
    if len(valid)<2:
        return 0.0
    difference=np.diff(valid)
    return 100.0*float(np.sum(difference>0))/len(difference)

def two_ray_pattern(pos_cm,start_extra_cm=0.0):
    wavelength_cm=WAVELENGTH_M*100.0
    extra_path=start_extra_cm+2.0*pos_cm
    phase=2*np.pi*extra_path/wavelength_cm
    amplitude=np.abs(1.0+np.exp(1j*phase))
    signal_db=20.0*np.log10(np.clip(amplitude/2.0,1e-3,1.0))
    return signal_db

def print_report(pos,indoor,outdoor,time_s,stationary_rssi):
    indoor_roughness=roughness(indoor)
    indoor_fade=fade_depth(indoor)
    indoor_reversals=reversals(indoor)
    outdoor_roughness=roughness(outdoor)
    outdoor_fade=fade_depth(outdoor)
    outdoor_reversals=reversals(outdoor)
    wavelength_cm=WAVELENGTH_M*100
    half_wavelength_cm=wavelength_cm/2

    print("\n")
    print("="*75)
    print("="*75)
    print()
    print(f"Wavelength at 2.4 GHz = {wavelength_cm:.1f} cm")
    print(f"Peaks/nulls repeat approximately every {half_wavelength_cm:.1f} cm")
    print()
    print(f"{'Trace':<10} | {'Mean':<10} | {'Spread':<10} | {'Roughness':<12} | {'Fade Depth':<12} | {'Reversals'}")
    print("-"*75)

    print(f"{'Indoor':<10} | {np.mean(indoor):<10.1f} | {np.std(indoor):<10.1f} | {indoor_roughness:<12.1f} | {indoor_fade:<12.1f} | {indoor_reversals:.0f}%")
    print(f"{'Outdoor':<10} | {np.nanmean(outdoor):<10.1f} | {np.nanstd(outdoor):<10.1f} | {outdoor_roughness:<12.1f} | {outdoor_fade:<12.1f} | {outdoor_reversals:.0f}%")
    print("-"*75)

    stationary_mean=np.mean(stationary_rssi)
    stationary_spread=stationary_rssi.max()-stationary_rssi.min()

    print()
    print(f"Standing still: mean {stationary_mean:.1f} dBm")
    print(f"RSSI variation while stationary = {stationary_spread:.0f} dB")
    print("This variation can occur because the surrounding reflections change.")
    print()

    if outdoor_roughness>0:
        ratio=indoor_roughness/outdoor_roughness
    else:
        ratio=0

    scan_distance=pos[-1]-pos[0]

    print(f"Indoor trace is approximately {ratio:.1f} times rougher than outdoor.")
    print(f"Indoor peak-to-null swing = {indoor_fade:.0f} dB")
    print(f"Scan distance = {scan_distance:.0f} cm")
    print(f"RSSI became stronger on {indoor_reversals:.0f}% of indoor steps.")
    print()
    print("This variation demonstrates the effect of multipath fading.")
    print()
    print("="*75)
    print("END OF REPORT")
    print("="*75)
    print()

def plot_all(pos,indoor,outdoor,time_s,stationary_rssi):
    fig,axes=plt.subplots(2,2,figsize=(14,9))

    axes[0,0].plot(pos,indoor,'o-',color='#3b3b3b',linewidth=2,markersize=6,label='Indoor')
    axes[0,0].plot(pos,outdoor,'s--',color='#7a7a7a',linewidth=2,markersize=5,label='Outdoor')
    axes[0,0].set_title("Multipath Fading: RSSI vs Position")
    axes[0,0].set_xlabel("Position (cm)")
    axes[0,0].set_ylabel("RSSI (dBm)")
    axes[0,0].legend()
    axes[0,0].grid(True,linestyle=':',alpha=0.6)

    axes[0,1].plot(time_s,stationary_rssi,'o-',color='#5c5c5c',linewidth=2)
    mean_value=np.mean(stationary_rssi)
    axes[0,1].axhline(mean_value,color='gray',linestyle=':',linewidth=2,label=f"Mean = {mean_value:.1f} dBm")
    spread=stationary_rssi.max()-stationary_rssi.min()
    axes[0,1].set_title(f"Stationary RSSI Variation = {spread:.0f} dB")
    axes[0,1].set_xlabel("Time (s)")
    axes[0,1].set_ylabel("RSSI (dBm)")
    axes[0,1].legend()
    axes[0,1].grid(True,linestyle=':',alpha=0.6)

    labels=["Indoor","Outdoor"]
    rough_values=[roughness(indoor),roughness(outdoor)]
    fade_values=[fade_depth(indoor),fade_depth(outdoor)]
    x=np.arange(len(labels))
    width=0.35

    axes[1,0].bar(x-width/2,rough_values,width,label="Roughness")
    axes[1,0].bar(x+width/2,fade_values,width,label="Fade Depth")

    for i in range(len(labels)):
        axes[1,0].text(x[i]-width/2,rough_values[i]+0.3,f"{rough_values[i]:.1f}",ha="center")
        axes[1,0].text(x[i]+width/2,fade_values[i]+0.3,f"{fade_values[i]:.1f}",ha="center")

    axes[1,0].set_title("Indoor vs Outdoor Signal Variation")
    axes[1,0].set_ylabel("dB")
    axes[1,0].set_xticks(x)
    axes[1,0].set_xticklabels(labels)
    axes[1,0].legend()
    axes[1,0].grid(axis='y',linestyle=':',alpha=0.6)

    two_ray_position=np.linspace(0,50,500)
    signal_db=two_ray_pattern(two_ray_position)

    axes[1,1].plot(two_ray_position,signal_db,color='#5c5c5c',linewidth=2)
    axes[1,1].set_title("Two-Ray Theory: Peaks and Nulls")
    axes[1,1].set_xlabel("Receiver Position (cm)")
    axes[1,1].set_ylabel("Signal (dB relative to peak)")
    axes[1,1].set_ylim(-30,2)
    axes[1,1].grid(True,linestyle=':',alpha=0.6)

    fig.suptitle("Multipath Effects - Signal Fluctuation Observation",fontsize=17,fontweight="bold")
    plt.tight_layout(rect=[0,0,1,0.95])

    plt.savefig("multipath_all_graphs.png",dpi=150,bbox_inches="tight")
    print("Graph saved as: multipath_all_graphs.png")
    plt.show()

if __name__=="__main__":
    print("\nProgram started...\n")

    pos,indoor,outdoor=load_scan()
    time_s,stationary_rssi=load_stationary()

    print_report(pos,indoor,outdoor,time_s,stationary_rssi)

    plot_all(pos,indoor,outdoor,time_s,stationary_rssi)

    print("\nProgram finished successfully.")
